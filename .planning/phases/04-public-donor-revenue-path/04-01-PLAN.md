```typescript
"use client";
import { useEffect, useState } from "react";
import { api } from "@/lib/api";
import { ProgressBar } from "@/components/ui";
import { formatPrice } from "@/lib/format";

interface ProgressPollProps {
  slug: string;
  initialTotalRaised: number;
  initialDonorCount: number;
  goalAmount: number;
  hideAmount: boolean;
  hideDonors: boolean;
}

export function ProgressPoll({ slug, initialTotalRaised, initialDonorCount, goalAmount, hideAmount, hideDonors }: ProgressPollProps) {
  const [totalRaised, setTotalRaised] = useState(initialTotalRaised);
  const [donorCount, setDonorCount] = useState(initialDonorCount);

  useEffect(() => {
    let cancelled = false;
    const tick = async () => {
      if (document.visibilityState !== "visible") return;
      try {
        const fresh = await api<{ totalRaised: number; donorCount: number }>(`/api/cagnottes/${slug}`);
        if (!cancelled && fresh) {
          setTotalRaised(fresh.totalRaised);
          setDonorCount(fresh.donorCount);
        }
      } catch { /* silent */ }
    };
    const interval = setInterval(tick, 20_000);
    const onVis = () => { if (document.visibilityState === "visible") tick(); };
    document.addEventListener("visibilitychange", onVis);
    return () => { cancelled = true; clearInterval(interval); document.removeEventListener("visibilitychange", onVis); };
  }, [slug]);

  const percent = goalAmount > 0 ? Math.min(100, Math.round((totalRaised / goalAmount) * 100)) : 0;

  return (
    <div className="space-y-2">
      {!hideAmount && (
        <div className="font-headings text-2xl font-bold text-primary">
          {formatPrice(totalRaised)} <span className="text-sm font-normal text-muted-foreground">/ {formatPrice(goalAmount)}</span>
        </div>
      )}
      <ProgressBar value={percent} max={100} />
      {!hideDonors && (
        <p className="text-sm text-muted-foreground">{donorCount} {donorCount === 1 ? "participant" : "participants"}</p>
      )}
    </div>
  );
}
```

```typescript
"use client";
import { useEffect, useState } from "react";
import { useParams, useSearchParams } from "next/navigation";
import Link from "next/link";
import { api } from "@/lib/api";
import { Button } from "@/components/ui";
import { MiniCagnotteCard } from "@/components/checkout/MiniCagnotteCard";
import { ShareSheet } from "@/components/share/ShareSheet";
import { formatPrice } from "@/lib/format";
import { MERCI_LABELS } from "@/lib/constants";

const POLL_INTERVAL_MS = 3000;
const MAX_POLLS = 40; // 3s × 40 = 2min
const BASE_URL = process.env.NEXT_PUBLIC_BASE_URL || "https://cagnottes.sn";

type Status = "PENDING" | "PAID" | "FAILED" | "EXPIRED" | "TIMEOUT";

export default function MerciPage() {
  const params = useParams<{ slug: string }>();
  const searchParams = useSearchParams();
  const refFromUrl = searchParams.get("ref");
  const slug = params.slug;

  const [reference, setReference] = useState<string | null>(refFromUrl);
  const [status, setStatus] = useState<Status>("PENDING");
  const [order, setOrder] = useState<any | null>(null);
  const [attempts, setAttempts] = useState(0);

  // Hydrate reference from sessionStorage if URL query missing (TikTok WebView edge case)
  useEffect(() => {
    if (!reference) {
      const stored = sessionStorage.getItem(`cagnotte.order.${slug}`);
      if (stored) setReference(stored);
    }
  }, [reference, slug]);

  useEffect(() => {
    if (!reference) return;
    if (status !== "PENDING") return;
    if (attempts >= MAX_POLLS) { setStatus("TIMEOUT"); return; }

    let cancelled = false;
    const id = setTimeout(async () => {
      if (document.visibilityState !== "visible") {
        // Pause polling when tab hidden — re-trigger via visibilitychange handler
        return;
      }
      try {
        const data = await api<{ status: Status; amount?: number; donorMessage?: string; thankYouMessage?: string; customerName?: string }>(`/api/orders/${reference}/status`);
        if (cancelled) return;
        setOrder(data);
        if (data.status === "PAID" || data.status === "FAILED" || data.status === "EXPIRED") {
          setStatus(data.status);
        } else {
          setAttempts((n) => n + 1);
        }
      } catch {
        if (!cancelled) setAttempts((n) => n + 1);
      }
    }, POLL_INTERVAL_MS);

    return () => { cancelled = true; clearTimeout(id); };
  }, [reference, status, attempts]);

  // Wake polling when tab becomes visible
  useEffect(() => {
    const handler = () => { if (document.visibilityState === "visible" && status === "PENDING") setAttempts((n) => n); };
    document.addEventListener("visibilitychange", handler);
    return () => document.removeEventListener("visibilitychange", handler);
  }, [status]);

  function manualRetry() { setStatus("PENDING"); setAttempts(0); }

  if (!reference) {
    return (
      <div className="container mx-auto p-8 text-center">
        <p>{MERCI_LABELS.headingFailed}</p>
        <Link href={`/c/${slug}`}><Button variant="outline">Retour à la cagnotte</Button></Link>
      </div>
    );
  }

  return (
    <div className="container mx-auto px-4 py-8 max-w-2xl text-center space-y-6">
      {status === "PENDING" && (
        <>
          <div className="text-4xl">⏳</div>
          <h1 className="font-headings text-2xl font-bold">{MERCI_LABELS.headingPending}</h1>
          <p className="text-sm text-muted-foreground">Tentative {attempts + 1}/{MAX_POLLS}</p>
        </>
      )}

      {status === "PAID" && (
        <>
          <div className="text-5xl">✓</div>
          <h1 className="font-headings text-3xl font-bold text-primary">{MERCI_LABELS.headingPaid}</h1>
          {order?.amount && (
```

## Trust boundaries

| Boundary | Description |
|----------|-------------|
| public donor (anonymous, no auth) → `POST /api/orders` | Untrusted form input crosses into the backend. Backend enforces Zod validation, rate limits, circuit breaker. |
| public donor → `GET /api/cagnottes/*` | Untrusted slug parameter; backend handles 404 + visibility filter. |
| public donor → `/api/pay-redirect?t=<base64>` | Untrusted base64 query; backend validates against allowlist. (Sealed route — Phase 4 does not modify.) |
| Bictorys WebView → `/c/{slug}/merci?ref=...` | Bictorys is the source of truth for `ref`; if attacker forges a `ref`, the status endpoint returns 404 / unrelated order — no privilege escalation since order data is non-sensitive. |
| client-side `sessionStorage` ↔ `/paiement` | Untrusted local storage; must NOT be relied on for security decisions. We only use it for UX continuity. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Tampering | `POST /api/orders` body from public donor | mitigate | Backend Zod schema validates `amount` (500-10M FCFA), `paymentType` enum, length limits on `customerName`/`donorMessage`. **No frontend Zod mirror** — we trust backend errors. (Already in place from Phase 2.) |

| T-04-03 | Information Disclosure | Private cagnotte SEO leak (P05) | mitigate | Three layers: (a) `robots.txt` disallows `/c/`, (b) `generateMetadata` sets `robots: { index: false, follow: false }` per page, (c) `force-dynamic` on `/c/[slug]` so no static pages are built at CI time. T3 task. |
| T-04-04 | Information Disclosure | Donor email harvesting via participants list | mitigate | `GET /api/cagnottes/:slug/participants` masks anonymous donors and never returns `customerEmail`. (Backend Phase 2 — verified.) Phase 4 only consumes the masked output. |

| T-04-07 | Denial of Service | Spam donations via /paiement POST | mitigate | Backend rate limits: 20/min/IP + 100/hr/IP + 5/min/email + Bictorys circuit breaker (5 failures/30s → 60s cooldown). Phase 4 surfaces 429/503 errors as user-friendly toast (T5). |

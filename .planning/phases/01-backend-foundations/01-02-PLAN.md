```typescript
/**
 * Cagnotte slug helpers — Phase 1 plan 01-02.
 *
 * slugify() is pure and deterministic.
 * ensureUniqueSlug() takes a closure so the helper stays testable without a real Prisma client.
 *
 * Pitfalls guarded:
 *   P04 — slug reservation race → the unique index from plan 01-01 is the backstop; this retries on P2002.
 *   P12 — diacritic edge cases → NFD normalize + combining-marks strip covers all French accents.
 *
 * Per PROJECT.md: slugs are human-readable with numeric suffixes. NO random hex.
 */

import { Prisma } from "../../generated/prisma/client.js";

/**
 * Reserved slugs for cagnotte blocks (DISTINCT from Seller.slug RESERVED_SLUGS in constants.ts).
 * A cagnotte cannot be created at cagnottes.sn/c/admin, etc.
 * LOCKED — do not add or remove without updating RESEARCH.md constraint 5.
 */
export const BLOCK_RESERVED_SLUGS: ReadonlySet<string> = new Set([
  "api",
  "admin",
  "login",
  "signup",
  "dashboard",
  "tableau-de-bord",
  "profil",
  "notifications",
  "participations",
  "aide",
  "tarifs",
  "contact",
  "nouvelle",
  "create",
  "toutes-les-cagnottes",
]);

/**
 * Convert a French title into a URL-safe slug.
 *
 *   "Les 30 ans de Thomas Diémé" → "les-30-ans-de-thomas-dieme"
 *   "Cadeau pour Aïssata"        → "cadeau-pour-aissata"
 *   ""                           → "cagnotte" (empty fallback)
 *   "   !!!   "                  → "cagnotte"
 *
 * Rules:
 *  1. NFD normalize + strip combining marks (diacritic removal)
 *  2. lowercase
 *  3. collapse any run of non-[a-z0-9] to a single dash
 *  4. trim leading/trailing dashes
 *  5. truncate to 60 chars, then re-trim trailing dash
 *  6. if empty after all of the above, return "cagnotte"
 */
export function slugify(title: string): string {
  const slug = title
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")   // strip combining marks (é→e, ï→i, …)
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")       // collapse to single dashes
    .replace(/^-+|-+$/g, "")           // trim leading/trailing dashes
    .slice(0, 60)
    .replace(/-+$/, "");               // re-trim after truncate

  return slug || "cagnotte";
}

/**
 * Type guard for P2002 — works for both real Prisma errors and test-harness fakes.
 * Real errors satisfy `instanceof`; test fakes satisfy the `.code === "P2002"` duck check.
 */
function isUniqueConstraintError(err: unknown): boolean {
  if (err instanceof Prisma.PrismaClientKnownRequestError && err.code === "P2002") {
    return true;
  }
  // Duck-typed fallback: test harness + cases where Prisma loses the prototype chain.
  return (
    typeof err === "object" &&
    err !== null &&
    "code" in err &&
    (err as { code?: unknown }).code === "P2002"
  );
}

/**
 * Attempt to reserve `base` as a slug by calling createFn. On P2002, retry with
 * `base-2` through `base-10`, then fall back to a 4-char timestamp suffix.
 *
 * `base` is assumed to already be the output of slugify() — callers are responsible
 * for normalization. Passing a non-slug string is undefined behavior.
 *
 * If `base` is in BLOCK_RESERVED_SLUGS, the first attempt is `${base}-1` (NOT `base`).
 */
export async function ensureUniqueSlug(
  base: string,
  createFn: (candidate: string) => Promise<void>,
): Promise<string> {
  const start = BLOCK_RESERVED_SLUGS.has(base) ? `${base}-1` : base;

  // Attempt 1 — start candidate
  try {
    await createFn(start);
    return start;
  } catch (err) {
    if (!isUniqueConstraintError(err)) throw err;
  }

  // Attempts 2..10 — numeric suffix
  for (let n = 2; n <= 10; n++) {
    const candidate = `${base}-${n}`;
    try {
      await createFn(candidate);
      return candidate;
    } catch (err) {
      if (!isUniqueConstraintError(err)) throw err;
    }
  }

  // Fallback — 4-char timestamp suffix (NOT random hex, per PROJECT.md decision).
  const fallback = `${base}-${Date.now().toString(36).slice(-4)}`;
  await createFn(fallback); // if THIS throws P2002, bubble up — astronomical collision.
  return fallback;
}
```

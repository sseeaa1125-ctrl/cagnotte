    import { Router } from "express";
    import { z } from "zod";
    import { prisma } from "../lib/prisma.js";
    import { requireAuth } from "../middleware/auth.js";

    export const notificationsRouter = Router();
    notificationsRouter.use(requireAuth);

    notificationsRouter.get("/", async (req, res) => {
      const cursor = (req.query.cursor as string) || undefined;
      const limit = Math.min(parseInt((req.query.limit as string) || "20", 10), 100);
      const items = await prisma.notification.findMany({
        where: { sellerId: req.user!.id },
        orderBy: { createdAt: "desc" },
        take: limit + 1,
        ...(cursor ? { cursor: { id: cursor }, skip: 1 } : {}),
      });
      const hasMore = items.length > limit;
      const nextCursor = hasMore ? items[limit - 1].id : null;
      res.json({ items: items.slice(0, limit), nextCursor });
    });

    notificationsRouter.get("/count", async (req, res) => {
      const sellerId = req.user!.id;
      const [total, unread] = await Promise.all([
        prisma.notification.count({ where: { sellerId } }),
        prisma.notification.count({ where: { sellerId, isRead: false } }),
      ]);
      res.json({ total, unread });
    });

    notificationsRouter.post("/mark-read", async (req, res) => {
      const schema = z.object({ ids: z.array(z.string()).optional(), all: z.boolean().optional() });
      const data = schema.parse(req.body);
      const sellerId = req.user!.id;
      let updated;
      if (data.all) {
        updated = await prisma.notification.updateMany({
          where: { sellerId, isRead: false },
          data: { isRead: true },
        });
      } else if (data.ids && data.ids.length > 0) {
        updated = await prisma.notification.updateMany({
          where: { sellerId, id: { in: data.ids }, isRead: false },
          data: { isRead: true },
        });
      } else {
        res.status(400).json({ error: "ids[] ou all=true requis" });
        return;
      }
      res.json({ updated: updated.count });
    });

    notificationsRouter.get("/prefs", async (req, res) => {
      const seller = await prisma.seller.findUnique({
        where: { id: req.user!.id },
        select: { notificationPrefs: true },
      });
      res.json(seller?.notificationPrefs || {});
    });

    notificationsRouter.patch("/prefs", async (req, res) => {

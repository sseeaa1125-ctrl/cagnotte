    ```typescript
    /**
     * Manual KYC approval/rejection helper for v1 (no admin panel yet).
     *
     * Usage:
     *   cd backend && npx tsx scripts/approve-kyc.ts <seller-slug> [APPROVED|REJECTED] [reason]
     *
     * Default status: APPROVED.
     * Fires KYC_APPROVED / KYC_REJECTED notification via createNotification.
     */
    import { prisma } from "../src/lib/prisma.js";
    import { fireKycApproved, fireKycRejected } from "../src/lib/notifications/dispatch.js";

    async function main() {
      const [, , slug, statusArg, ...reasonParts] = process.argv;
      if (!slug) {
        console.error("Usage: tsx scripts/approve-kyc.ts <seller-slug> [APPROVED|REJECTED] [reason]");
        process.exit(1);
      }
      const status = (statusArg || "APPROVED").toUpperCase() as "APPROVED" | "REJECTED";
      if (status !== "APPROVED" && status !== "REJECTED") {
        console.error(`Invalid status: ${status}. Use APPROVED or REJECTED.`);
        process.exit(1);
      }
      const reason = reasonParts.join(" ") || "Documents non conformes";

      const seller = await prisma.seller.findUnique({ where: { slug } });
      if (!seller) {
        console.error(`Seller not found: ${slug}`);
        process.exit(1);
      }

      await prisma.seller.update({
        where: { id: seller.id },
        data: { kycStatus: status },
      });

      if (status === "APPROVED") {
        const result = await fireKycApproved(seller);
        console.log(`✓ Seller ${slug} → APPROVED. Notification: ${result.created ? "fired" : "deduped"}.`);
      } else {
        const result = await fireKycRejected(seller, reason);
        console.log(`✓ Seller ${slug} → REJECTED. Reason: ${reason}. Notification: ${result.created ? "fired" : "deduped"}.`);
      }
      await prisma.$disconnect();
    }

    main().catch((err) => {
      console.error(err);
      process.exit(1);
    });
    ```

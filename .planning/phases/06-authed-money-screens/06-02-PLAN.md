**0a. `src/lib/withdrawal/schema.ts`**

Client-side mirror of the backend withdrawal request shape. NO zod dep in the frontend (if not already installed — check first with `grep '"zod"' package.json`; do NOT add it if absent — use manual validators).

```typescript
export type PayoutProvider = "wave_money" | "orange_money";

export interface WithdrawalDraft {
  amount: number;           // FCFA integer
  provider: PayoutProvider;
  phone: string;            // +221 prefixed or 9-digit national
  recipientName: string;
  pin: string;              // 4 digits, entered in step 2
}

export const PIN_LENGTH = 4;
export const PIN_REGEX = /^\d{4}$/;

export function validateWithdrawalDraft(draft: Partial<WithdrawalDraft>, balance: number): string | null {
  if (!draft.amount || draft.amount <= 0) return "Montant invalide";
  if (draft.amount > balance) return "Montant supérieur au solde disponible";
  if (!draft.provider) return "Choisissez un moyen de paiement";
  if (!draft.phone || !draft.phone.trim()) return "Numéro manquant";
  if (!draft.recipientName || !draft.recipientName.trim()) return "Nom du bénéficiaire manquant";
  return null;
}

export function validatePin(pin: string): boolean {
  return pin.length === PIN_LENGTH && PIN_REGEX.test(pin);
}
```

**0b. `src/hooks/useWithdrawalDraft.ts`**

Copy the structure of `useWizardDraft.ts`. Key: `cagnotte.withdrawal.draft.v1`. sessionStorage, 24h staleness, SSR-safe `isReady` gate.

```typescript
"use client";
import { useEffect, useState } from "react";
import type { WithdrawalDraft } from "@/lib/withdrawal/schema";

const KEY = "cagnotte.withdrawal.draft.v1";
const MAX_AGE_MS = 24 * 60 * 60 * 1000;

interface Envelope { value: Partial<WithdrawalDraft>; savedAt: number; }

export function useWithdrawalDraft() {
  const [draft, setDraftState] = useState<Partial<WithdrawalDraft>>({});
  const [isReady, setIsReady] = useState(false);

  useEffect(() => {
    try {
      const raw = sessionStorage.getItem(KEY);
      if (raw) {
        const env = JSON.parse(raw) as Envelope;
        if (Date.now() - env.savedAt < MAX_AGE_MS) {
          setDraftState(env.value);
        } else {
          sessionStorage.removeItem(KEY);
        }
      }
    } catch {}
    setIsReady(true);
  }, []);

  const setDraft = (patch: Partial<WithdrawalDraft>) => {
    setDraftState(prev => {
      const next = { ...prev, ...patch };
      try { sessionStorage.setItem(KEY, JSON.stringify({ value: next, savedAt: Date.now() } satisfies Envelope)); } catch {}
      return next;
    });
  };

  const clear = () => {
    try { sessionStorage.removeItem(KEY); } catch {}
    setDraftState({});
  };

  return { draft, setDraft, clear, isReady };
}
```
